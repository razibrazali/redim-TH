async function submitParcel() { if (!input.value.trim() || !capturedPhotoData) { addLog('Submit attempted with missing data', 'error'); alert('Please complete all steps before submitting.'); return; } addLog('=== SUBMITTING PARCEL ===', 'info'); addLog('Tracking Number: ' + input.value.trim(), 'info'); addLog('Photo data size: ' + Math.round(capturedPhotoData.length / 1024) + ' KB', 'info'); submitButton.disabled = true; submitButton.textContent = '⏳ Processing...'; try { addLog('Sending request to Google Apps Script...', 'info'); addLog('URL: ' + GOOGLE_SCRIPT_URL, 'info'); const payload = { action: 'submitParcel', trackingNumber: input.value.trim(), photoBase64: capturedPhotoData }; addLog('Payload prepared', 'info'); // Use text/plain to avoid CORS preflight const response = await fetch(GOOGLE_SCRIPT_URL, { method: 'POST', mode: 'cors', headers: { 'Content-Type': 'text/plain;charset=utf-8', }, body: JSON.stringify(payload) }); addLog('Response received!', 'info'); addLog('Response status: ' + response.status, 'info'); const responseText = await response.text(); addLog('Response text length: ' + responseText.length + ' characters', 'info'); addLog('Response text preview: ' + responseText.substring(0, 200) + '...', 'info'); const result = JSON.parse(responseText); addLog('Response parsed successfully', 'info'); addLog('Success: ' + result.success, result.success ? 'info' : 'error'); addLog('Message: ' + result.message, result.success ? 'info' : 'error'); // Display backend logs if (result.logs && result.logs.length > 0) { addLog('=== BACKEND LOGS ===', 'warning'); result.logs.forEach(log => { addLog('[BACKEND] ' + log, 'info'); }); addLog('=== END BACKEND LOGS ===', 'warning'); } if (result.success && result.data) { addLog('=== SUBMISSION COMPLETE ===', 'info'); addLog('Final Tracking Number: ' + result.data.trackingNumber, 'info'); addLog('Final Weight: ' + result.data.weight, 'info'); addLog('Final Drive Link: ' + result.data.driveLink, 'info'); showSummary( result.data.trackingNumber, result.data.weight, result.data.driveLink ); submitButton.disabled = false; submitButton.textContent = '✅ Submit Parcel'; } else { addLog('Submission failed: ' + result.message, 'error'); alert('Error: ' + result.message); submitButton.disabled = false; submitButton.textContent = '✅ Submit Parcel'; } } catch (error) { console.error('Submit error:', error); addLog('Submit error: ' + error.message, 'error'); addLog('Error stack: ' + error.stack, 'error'); alert('Error submitting parcel: ' + error.message); submitButton.disabled = false; submitButton.textContent = '✅ Submit Parcel'; } }
